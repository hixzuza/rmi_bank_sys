public static void main(String[] args) throws Exception {

    DbManager db = new DbManager();
    Scanner scanner = new Scanner(System.in);

    // ════════════════════════════════════════
    // 1. SEED THE DB (run once, then comment out)
    // ════════════════════════════════════════
    // db.test_the_db();

    // ════════════════════════════════════════
    // 2. AUTH
    // ════════════════════════════════════════
    System.out.println("\n=== AUTH ===");
    System.out.println("client1 / client123 -> " + db.auth("client1", "client123")); // true
    System.out.println("client1 / wrong     -> " + db.auth("client1", "wrong"));     // false
    System.out.println("nobody / whatever   -> " + db.auth("nobody",  "whatever"));  // false
    System.out.println("admin1 / admin123   -> " + db.auth("admin1",  "admin123"));  // true

    // ════════════════════════════════════════
    // 3. CONSULTER SOLDE
    // ════════════════════════════════════════
    System.out.println("\n=== CONSULTER SOLDE ===");
    System.out.println("C001 solde = " + db.consulterSolde("C001")); // 1250.75
    System.out.println("C002 solde = " + db.consulterSolde("C002")); // 350.00
    System.out.println("FAKE solde = " + db.consulterSolde("FAKE")); // -1

    // ════════════════════════════════════════
    // 4. DEPOSER
    // ════════════════════════════════════════
    System.out.println("\n=== DEPOSER ===");
    boolean dep = db.deposer("C001", 200.0);
    db.enregistrerTransaction("C001", "DEPOT", 200.0, null);
    System.out.println("Depot 200 sur C001 -> " + dep);
    System.out.println("C001 nouveau solde = " + db.consulterSolde("C001")); // 1450.75

    // ════════════════════════════════════════
    // 5. RETIRER
    // ════════════════════════════════════════
    System.out.println("\n=== RETIRER ===");
    boolean ret = db.retirer("C001", 100.0);
    db.enregistrerTransaction("C001", "RETRAIT", 100.0, null);
    System.out.println("Retrait 100 sur C001 -> " + ret);
    System.out.println("C001 nouveau solde   = " + db.consulterSolde("C001")); // 1350.75

    boolean retFail = db.retirer("C002", 99999.0); // solde insuffisant
    System.out.println("Retrait 99999 sur C002 (doit echouer) -> " + retFail); // false

    // ════════════════════════════════════════
    // 6. VIREMENT
    // ════════════════════════════════════════
    System.out.println("\n=== VIREMENT ===");
    boolean vir = db.virement("C001", "C002", 50.0);
    db.enregistrerTransaction("C001", "VIREMENT", 50.0, "C002");
    System.out.println("Virement 50 C001->C002 -> " + vir);
    System.out.println("C001 solde = " + db.consulterSolde("C001")); // 1300.75
    System.out.println("C002 solde = " + db.consulterSolde("C002")); // 400.00

    // ════════════════════════════════════════
    // 7. HISTORIQUE
    // ════════════════════════════════════════
    System.out.println("\n=== HISTORIQUE C001 ===");
    db.getHistorique("C001").forEach(System.out::println);

    // ════════════════════════════════════════
    // 8. LISTER COMPTES PAR USER
    // ════════════════════════════════════════
    System.out.println("\n=== COMPTES USER id=1 ===");
    db.listerComptesParUser(1).forEach(System.out::println);

    // ════════════════════════════════════════
    // 9. ADMIN - LISTER TOUS LES COMPTES
    // ════════════════════════════════════════
    System.out.println("\n=== ALL COMPTES ===");
    db.listerComptes().forEach(System.out::println);

    // ════════════════════════════════════════
    // 10. ADMIN - CREER COMPTE
    // ════════════════════════════════════════
    System.out.println("\n=== CREER COMPTE ===");
    boolean created = db.creerCompte("client1", 999.0);
    System.out.println("Nouveau compte cree -> " + created); // true
    db.listerComptes().forEach(System.out::println);

    // ════════════════════════════════════════
    // 11. ADMIN - SUPPRIMER COMPTE
    // ════════════════════════════════════════
    System.out.println("\n=== SUPPRIMER COMPTE ===");
    boolean deleted = db.supprimerCompte("C003");
    System.out.println("C003 desactive -> " + deleted); // true
    System.out.println("C003 solde apres suppression = " + db.consulterSolde("C003")); // -1 (actif=false)

    scanner.close();
}
